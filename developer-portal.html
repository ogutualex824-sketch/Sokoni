Quickstart

Call your first Cloud Function in 5 minutes
1
Install Firebase SDK
Add to your HTML <head> or install via npm.
<!-- CDN --> <script src="https://www.gstatic.com/firebasejs/10.12.0/firebase-app-compat.js"></script> <script src="https://www.gstatic.com/firebasejs/10.12.0/firebase-auth-compat.js"></script> <script src="https://www.gstatic.com/firebasejs/10.12.0/firebase-functions-compat.js"></script> <script src="https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore-compat.js"></script> // npm npm install firebase
2
Load Config & Initialize
All apps use the same config from /sokoni-config.js.
<script src="/sokoni-config.js"></script> <!-- exposes window.SOKONI_CONFIG --> const app = firebase.initializeApp(window.SOKONI_CONFIG, 'my-app'); const auth = firebase.auth(app); const fns = firebase.functions(app); fns.useRegion('us-central1');
3
Authenticate
All callable functions require a Firebase Auth user. Read custom claims to check role.
auth.onAuthStateChanged(async user => { if (!user) { window.location.href = '/login.html'; return; } const token = await user.getIdTokenResult(); const role = token.claims.role || 0; // 0=guest 1=buyer 2=seller 3=dev 4=admin 5=super const shopId = token.claims.shopId; // set for sellers });
4
Call a Cloud Function
// Helper wrapper const callCF = async (name, data = {}) => { const r = await fns.httpsCallable(name)(data); return r.data; }; // Example: search products const results = await callCF('searchProducts', { query: 'maize flour', limit: 20 }); console.log(results.hits);
5
Handle Errors
try { const order = await callCF('createOrder', data); } catch (err) { switch (err.code) { case 'functions/unauthenticated': window.location.href = '/login.html'; break; case 'functions/permission-denied': alert('Access denied'); break; case 'functions/resource-exhausted': /* rate limit — back off and retry */ break; case 'functions/not-found': alert('Not found'); break; default: console.error(err.message); } }

Platform Events
Register a Webhook
await callCF('registerWebhook', { url: 'https://yoursite.com/hook', events: ['order.created', 'payment.success'], name: 'My Integration', });
Verify Webhook Signature (Node.js)
const crypto = require('crypto'); function verify(body, sig, secret) { const expected = crypto .createHmac('sha256', secret) .update(JSON.stringify(body)) .digest('hex'); return crypto.timingSafeEqual( Buffer.from(expected), Buffer.from(sig) ); } // Express middleware app.post('/hook', (req, res) => { const sig = req.headers['x-sokoni-signature']; if (!verify(req.body, sig, process.env.WEBHOOK_SECRET)) return res.status(401).end(); // Process event res.sendStatus(200); });
